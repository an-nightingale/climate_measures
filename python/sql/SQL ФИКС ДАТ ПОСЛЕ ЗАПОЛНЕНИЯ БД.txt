UPDATE period
SET
    start_date = CASE
        WHEN name ~ '^\d{4} год$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^январь-март \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^январь-июнь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^январь-сентябрь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^январь-декабрь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^I квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^II квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 4, 1)

        WHEN name ~ '^III квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 7, 1)

        WHEN name ~ '^IV квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 10, 1)

        WHEN name ~ '^на 1 января \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^на 1 апреля \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 4, 1)

        WHEN name ~ '^на 1 июля \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 7, 1)

        WHEN name ~ '^на 1 октября \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 10, 1)

        ELSE start_date
    END,

    end_date = CASE
        WHEN name ~ '^\d{4} год$'
            THEN make_date(substring(name from '(\d{4})')::int, 12, 31)

        WHEN name ~ '^январь-март \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 3, 31)

        WHEN name ~ '^январь-июнь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 6, 30)

        WHEN name ~ '^январь-сентябрь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 9, 30)

        WHEN name ~ '^январь-декабрь \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 12, 31)

        WHEN name ~ '^I квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 3, 31)

        WHEN name ~ '^II квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 6, 30)

        WHEN name ~ '^III квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 9, 30)

        WHEN name ~ '^IV квартал \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 12, 31)

        WHEN name ~ '^на 1 января \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 1, 1)

        WHEN name ~ '^на 1 апреля \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 4, 1)

        WHEN name ~ '^на 1 июля \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 7, 1)

        WHEN name ~ '^на 1 октября \d{4}$'
            THEN make_date(substring(name from '(\d{4})')::int, 10, 1)

        ELSE end_date
    END
WHERE start_date IS NULL
   OR end_date IS NULL;
